# 就餐项目实际开发

> 开发一：获取订阅信息接口  初步完成验证



> 定位：本层只做**回调数据落库**（事件流水，按事件逐条追加，同一单据多个审批节点多行，用 PROCESS_ID 归集）。 **不做**单据类型过滤、不做停餐/用餐业务判定——那些属于后续解析层。



> 全部保存入库，不在入口过滤类型：approvalType、settingName、processStatus 一律不筛，先存下来。原因：formInfo 字段名尚在确认中，入口筛错会永久丢数据（回调不重发）。
>
> 
>
> 不限制流程状态：报工单代码写死 processStatus=INITIATE && operatorStatus=approveInitiate，就餐项目必须放开，否则 PASS / WITHDRAW / DENIED 等事件全都收不到。（不加限制
>
> 
>
> **绝不因"非目标单据"返回失败**：IHR 规定同模板异常 10 次将熔断接口半小时。因此 Controller 恒返回 `{code:0, message:""}`，Service 内部异常全部 catch 后仅记日志。
>
> 
>
> **幂等去重**：IHR 失败会异步重试（`extraInfo.isRetry` / `retryCount`），同一事件会重复推送，入库前先查重。

@RestController
@RequestMapping("/api/hris")
public class HrisCallBackController extends BaseController {
    private final Logger logger = LoggerFactory.getLogger(HrisCallBackController.class);

    @Autowired
    public IHrisApiService hrisApiService;

   /**
     * IHR 发起和审批单据后同步数据（回调入口）
     * 现阶段：请求返回数据内容(就餐项目)
     *
     * 注意：
     * 1) 必须恒定返回 {"code":0,"message":""}。IHR 规定返回格式错误将导致同步异常，
     *    且同一模板异常10次会熔断接口半小时，因此无论业务处理结果如何，此处均返回成功。

     */
    @RequestMapping(value = "/v1/workflow/webhook/submit", method = RequestMethod.POST, produces = "application/json;charset=UTF-8")
    @HrisCallBackVerify
    public HashMap<String, Object> eventCall(@RequestBody(required = false) JSONObject request){
        logger.info("--------HRIS审批回调事件begin callback:{}" , request);
        HashMap<String, Object> result = new HashMap<String, Object>();
        result.put("code", 0);
        result.put("message", "");
        hrisApiService.newHandleEventCall(request);
        logger.info("--------HRIS审批回调事件end callback");
        return result;
    }


}

        HrisInsDetail detail = new HrisInsDetail();

        detail.setId(UUID.randomUUID().toString().replace("-", ""));

        detail.setCompanyId(data.getString("companyId"));
        detail.setApplicantId(data.getString("applicantId"));

        detail.setOperatorName(data.getString("operatorName"));

        detail.setProcessCode(data.getString("processCode"));
        detail.setProcessStatus(data.getString("processStatus"));
        detail.setApprovalType(data.getString("approvalType"));
        detail.setVacationId(data.getString("vacationId"));

        JSONObject formInfo = data.getJSONObject("formInfo");
        detail.setFormInfo(formInfo == null ? null : formInfo.toJSONString());

        JSONObject extraInfo = data.getJSONObject("extraInfo");
        detail.setExtraInfo(extraInfo == null ? null : extraInfo.toJSONString());

        detail.setCreateBy(data.getString("applicantId"));
        detail.setCreateTime(new Date());
        detail.setDel("0");

        return detail;
    }
}

        log.info("HRIS回调单据落库: 成功|processId={}, processCode={}, processStatus={}, operatorStatus={}, settingName={}",
                detail.getProcessId(), detail.getProcessCode(), detail.getProcessStatus(),
                detail.getOperatorStatus(), detail.getSettingName());
        return rows > 0;
    }

    @Override
    public boolean haveExistData(HrisInsDetail detail) {
        return insDetailMapper.countInsDetail(
                detail.getProcessId(),
                detail.getOperatorId(),
                detail.getOperatorStatus(),
                detail.getProcessStatus()) > 0;
    }

    @Override
    public List<HrisInsDetail> queryByProcessId(String processId) {
        return insDetailMapper.queryByProcessId(processId);
    }
}

@Service
public class HrisApiServiceImpl implements IHrisApiService {
    private final Logger logger = LoggerFactory.getLogger(HrisApiServiceImpl.class);
    
    /** EXTRA_INFO 为 VARCHAR2(4000)，按字节计 */
    private static final int EXTRA_INFO_MAX_BYTES = 4000;

    @Autowired
    private IHrisWorkDataService workDataService;
    
     /**
     * 就餐项目：IHR审批回调事件入库
     *
     * 说明：
     * 1) 本方法只负责"收得全、答得快"——全量落库，不按 approvalType / settingName / processStatus 过滤；
     *    单据类型识别、是否总部、是否生效、停餐判定等，全部交由后续解析层处理。
     * 2) 任何异常均在此吞掉并记录日志，绝不向上抛出：
     *    IHR 规定同一模板异常10次将熔断接口半小时，故回调必须恒定返回成功。
     */
    @Override
    public void newHandleEventCall(JSONObject data) {
        try {
            if (data == null || data.isEmpty()) {
                logger.warn("HRIS回调事件: 忽略|回调数据为空");
                return;
            }

            // 仅校验归集所必需的最小字段，其余一律宽进
            String processId = data.getString("processId");
            if (StringUtils.isBlank(processId)) {
                logger.warn("HRIS回调事件: 忽略|缺少processId, data={}", data.toJSONString());
                return;
            }

            HrisInsDetail detail = HrisInsDetail.coverData(data);

            // EXTRA_INFO 超长保护：宁可截断并告警，也不要整条数据插入失败而丢事件
            detail.setExtraInfo(truncateByBytes(detail.getExtraInfo(), EXTRA_INFO_MAX_BYTES, processId));

            insDetailService.saveInsDetail(detail);

        } catch (Exception e) {
            logger.error("HRIS回调事件: 异常|入库失败, msg={}, data={}",
                    e.getMessage(), data == null ? "" : data.toJSONString(), e);
        }
    }

    /**
     * 按UTF-8字节长度安全截断，避免乱码
     */
    private String truncateByBytes(String value, int maxBytes, String processId) {
        if (StringUtils.isBlank(value)) {
            return value;
        }
        try {
            byte[] bytes = value.getBytes("UTF-8");
            if (bytes.length <= maxBytes) {
                return value;
            }
            logger.warn("HRIS回调事件: 警告|EXTRA_INFO超长已截断 processId={}, 原始字节={}", processId, bytes.length);
            int end = maxBytes;
            // 回退到完整字符边界（UTF-8续字节为 10xxxxxx）
            while (end > 0 && (bytes[end] & 0xC0) == 0x80) {
                end--;
            }
            return new String(bytes, 0, end, "UTF-8");
        } catch (UnsupportedEncodingException e) {
            logger.error("HRIS回调事件: 异常|截断EXTRA_INFO失败 processId={}", processId, e);
            return value.substring(0, Math.min(value.length(), maxBytes / 3));
        }
    }

}
